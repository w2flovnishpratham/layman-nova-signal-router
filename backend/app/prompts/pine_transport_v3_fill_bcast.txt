// ADMIN-ONLY SCRIPT. Do not distribute this source or the secret configured
// below to any subscriber -- this single chart is the sole signal source for
// every user who has selected this strategy in NOVA.
const string novaTransportVersion = "pine_transport_v3_fill_bcast"
const string novaSecretPlaceholder = "REPLACE_WITH_NOVA_MANAGED_SECRET"

novaSecret = input.string(novaSecretPlaceholder, "NOVA Managed Secret (admin-only)", group="NOVA Webhook")
novaSendHoldTest = input.bool(false, "Send one HOLD connectivity test", group="NOVA Webhook",
 tooltip="Connectivity only. Disable after one confirmed HOLD alert.")

bool novaSecretReady = novaSecret != novaSecretPlaceholder and str.length(novaSecret) >= 20

novaWebhookPayload(string action, string orderId) =>
    string novaSignalId = "NOVA-BROADCAST:" + syminfo.ticker + ":" + orderId + ":" + str.tostring(time) + ":" + action
    string novaSignalTime = str.format_time(time, "yyyy-MM-dd'T'HH:mm:ss'Z'", "UTC")
    '{"secret":"' + novaSecret + '","action":"' + action + '","signal_id":"' + novaSignalId + '","signal_time":"' + novaSignalTime + '","order_id":"' + orderId + '"}'

// Order-fill actions are reported by the alert_message= arguments NOVA
// attached to the preserved strategy order calls above, fired by TradingView
// itself on "Order fills only" alerts -- this block never calls alert()
// for BUY_CE/BUY_PE/EXIT. The HOLD connectivity test below is the only
// alert() this transport fires directly, so an admin can verify the webhook
// and secret before enabling live order-fill alerts.
bool novaAlertReady = barstate.isrealtime and barstate.isconfirmed and novaSecretReady
var bool novaHoldSent = false
if not novaSendHoldTest
    novaHoldSent := false
if novaAlertReady and novaSendHoldTest and not novaHoldSent
    alert(novaWebhookPayload("HOLD", "HOLD_TEST"), alert.freq_once_per_bar_close)
    novaHoldSent := true
// === NOVA FROZEN TRANSPORT END: pine_transport_v3_fill_bcast ===

// ADMIN-ONLY SCRIPT. Do not distribute this source or the secret configured
// below to any subscriber -- this single chart is the sole signal source for
// every user who has selected this strategy in NOVA.
const string novaTransportVersion = "pine_transport_v2_bcast"
const string novaSecretPlaceholder = "REPLACE_WITH_NOVA_MANAGED_SECRET"

novaSecret = input.string(novaSecretPlaceholder, "NOVA Managed Secret (admin-only)", group="NOVA Webhook")
novaSendHoldTest = input.bool(false, "Send one HOLD connectivity test", group="NOVA Webhook",
 tooltip="Connectivity only. Disable after one confirmed HOLD alert.")

bool novaSecretReady = novaSecret != novaSecretPlaceholder and str.length(novaSecret) >= 20

novaSignalId(string action) =>
    "NOVA-BROADCAST:" + syminfo.ticker + ":" + str.tostring(time_close) + ":" + action

novaSignalTime() =>
    str.format_time(time_close, "yyyy-MM-dd'T'HH:mm:ss'Z'", "UTC")

novaWebhookPayload(string action) =>
    '{"secret":"' + novaSecret + '","action":"' + action + '","signal_id":"' + novaSignalId(action) + '","signal_time":"' + novaSignalTime() + '"}'

bool novaAlertReady = barstate.isrealtime and barstate.isconfirmed and novaSecretReady

var bool novaHoldSent = false
if not novaSendHoldTest
    novaHoldSent := false
if novaAlertReady
    if novaSendHoldTest
        if not novaHoldSent
            alert(novaWebhookPayload("HOLD"), alert.freq_once_per_bar_close)
            novaHoldSent := true
    else
        if novaExitSignal
            alert(novaWebhookPayload("EXIT"), alert.freq_once_per_bar_close)
        else if novaBuyCeSignal
            alert(novaWebhookPayload("BUY_CE"), alert.freq_once_per_bar_close)
        else if novaBuyPeSignal
            alert(novaWebhookPayload("BUY_PE"), alert.freq_once_per_bar_close)
// === NOVA FROZEN TRANSPORT END: pine_transport_v2_bcast ===
